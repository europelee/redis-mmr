# 事件循环模型

Redis 采用基于 Reactor 模式的事件驱动架构，实现了高性能的单线程事件处理模型，并在后续版本中引入了 IO 多线程支持。

## 架构概览

```
+---------------------------------------------------------------+
|                    Redis 事件驱动架构                          │
+---------------------------------------------------------------+
|                                                               │
│   +------------------+                                       │
│   │   文件事件       │  File Events (Socket/Timer)          │
│   │   (File Events)  │                                       │
│   +--------+---------+                                       │
│            |                                                  │
│            v                                                  │
│   +--------+---------+                                       │
│   │   事件多路复用    │  epoll/kqueue/select/evport          │
│   │   (Multiplexer)  │                                       │
│   +--------+---------+                                       │
│            |                                                  │
│            v                                                  │
│   +--------+---------+       +------------------+            │
│   │   事件循环       |------>|   时间事件       │            │
│   │   (EventLoop)   │       │   (Time Events)  │            │
│   +--------+---------+       +------------------+            │
│            |                                                  │
│            v                                                  │
│   +--------+---------+                                       │
│   │   回调处理       │   Command Handlers / I/O Handlers    │
│   │   (Callbacks)    │                                       │
│   +------------------+                                       │
│                                                               │
+---------------------------------------------------------------+
```

## 1. 事件循环核心

### 1.1 数据结构

```c
/* 事件循环主结构 */
typedef struct aeEventLoop {
    int maxfd;                    /* 当前注册的最大文件描述符 */
    int setsize;                  /* 跟踪的集合大小 */
    long long timeEventNextId;    /* 下一个时间事件 ID */
    int nevents;                  /* 注册的事件数量 */

    aeFileEvent *events;          /* 文件事件数组 */
    aeFiredEvent *fired;          /* 已触发的事件数组 */
    aeTimeEvent *timeEventHead;   /* 时间事件链表头 */

    int stop;                     /* 停止标志 */
    void *apidata;                /* API 特定数据 (如 epoll_data) */

    /* 钩子函数 */
    aeBeforeSleepProc *beforesleep;  /* 睡眠前回调 */
    aeBeforeSleepProc *aftersleep;   /* 睡眠后回调 */

    int flags;                    /* 循环标志 */
    void *privdata[2];            /* 私有数据 */
} aeEventLoop;

/* 文件事件 */
typedef struct aeFileEvent {
    int mask;                     /* 事件掩码 (READABLE/WRITABLE/BARRIER) */
    aeFileProc *rfileProc;        /* 可读回调 */
    aeFileProc *wfileProc;        /* 可写回调 */
    void *clientData;             /* 回调参数 */
} aeFileEvent;

/* 时间事件 */
typedef struct aeTimeEvent {
    long long id;                 /* 事件 ID */
    monotime when;                /* 触发时间 */
    aeTimeProc *timeProc;         /* 回调函数 */
    aeEventFinalizerProc *finalizerProc;  /* 清理函数 */
    void *clientData;             /* 回调参数 */
    struct aeTimeEvent *prev;     /* 前向指针 */
    struct aeTimeEvent *next;     /* 后向指针 */
    int refcount;                 /* 引用计数 (防止递归释放) */
} aeTimeEvent;
```

### 1.2 事件类型

```c
/* 文件事件掩码 */
#define AE_NONE 0         /* 未注册 */
#define AE_READABLE 1     /* 可读事件 */
#define AE_WRITABLE 2     /* 可写事件 */
#define AE_BARRIER 4      /* 屏障模式 (先读后写) */

/* 事件循环标志 */
#define AE_FILE_EVENTS (1<<0)     /* 处理文件事件 */
#define AE_TIME_EVENTS (1<<1)     /* 处理时间事件 */
#define AE_ALL_EVENTS (AE_FILE_EVENTS|AE_TIME_EVENTS)
#define AE_DONT_WAIT (1<<2)       /* 不阻塞等待 */
#define AE_CALL_BEFORE_SLEEP (1<<3)  /* 调用睡眠前回调 */
#define AE_CALL_AFTER_SLEEP (1<<4)   /* 调用睡眠后回调 */
```

### 1.3 主循环逻辑

```c
/* 事件循环主函数 */
void aeMain(aeEventLoop *eventLoop) {
    eventLoop->stop = 0;
    while (!eventLoop->stop) {
        aeProcessEvents(eventLoop, AE_ALL_EVENTS);
    }
}

/* 处理事件的核心函数 */
int aeProcessEvents(aeEventLoop *eventLoop, int flags) {
    int processed = 0;

    /* 1. 处理时间事件 */
    if (flags & AE_TIME_EVENTS) {
        processed += aeProcessTimeEvents(eventLoop);
    }

    /* 2. 计算等待时间 */
    maxfd = -1;
    if (flags & AE_FILE_EVENTS) {
        /* 获取最近的时间事件 */
        aeTimeEvent *shortest = aeSearchNearestTimer(eventLoop);
        if (shortest) {
            now = getMonotonicUs();
            us = shortest->when - now;
            if (us <= 0) {
                /* 时间事件已到，立即处理 */
                tv.tv_sec = 0;
                tv.tv_usec = 0;
            } else {
                /* 计算等待时间 */
                tv.tv_sec = us / 1000000;
                tv.tv_usec = us % 1000000;
            }
        }
    }

    /* 3. 多路复用等待 */
    if (flags & AE_FILE_EVENTS) {
        numevents = aeApiPoll(eventLoop, &tv);

        /* 4. 处理已触发的事件 */
        for (j = 0; j < numevents; j++) {
            aeFiredEvent *fe = &eventLoop->fired[j];
            aeFileEvent *fev = &eventLoop->events[fe->fd];

            /* 处理可读事件 */
            if (fe->mask & AE_READABLE && fev->mask & AE_READABLE) {
                fev->rfileProc(eventLoop, fe->fd, fev->clientData, fe->mask);
            }

            /* 处理可写事件 */
            if (fe->mask & AE_WRITABLE && fev->mask & AE_WRITABLE) {
                if (!barrier_hit || fev->mask & AE_BARRIER) {
                    fev->wfileProc(eventLoop, fe->fd, fev->clientData, fe->mask);
                }
            }
        }
    }

    /* 5. 调用钩子 */
    if (eventLoop->aftersleep && (flags & AE_CALL_AFTER_SLEEP)) {
        eventLoop->aftersleep(eventLoop);
    }

    return processed;
}
```

### 2.2 epoll 实现

```c
/* epoll API 结构 */
static void aeApiAddEvent(aeEventLoop *eventLoop, int fd, int mask) {
    struct epoll_event ee;
    aeApiState *state = eventLoop->apidata;

    ee.events = 0;
    if (mask & AE_READABLE) ee.events |= EPOLLIN;
    if (mask & AE_WRITABLE) ee.events |= EPOLLOUT;

    ee.data.fd = fd;

    if (!update) {
        epoll_ctl(state->epfd, EPOLL_CTL_ADD, fd, &ee);
    } else {
        epoll_ctl(state->epfd, EPOLL_CTL_MOD, fd, &ee);
    }
}

static int aeApiPoll(aeEventLoop *eventLoop, struct timeval *tvp) {
    aeApiState *state = eventLoop->apidata;
    int retval, numevents = 0;

    retval = epoll_wait(state->epfd, state->events,
                        eventLoop->setsize,
                        tvp ? (tvp->tv_sec * 1000 + tvp->tv_usec / 1000) : -1);

    if (retval > 0) {
        numevents = retval;
        for (i = 0; i < numevents; i++) {
            int mask = 0;
            if (state->events[i].events & EPOLLIN) mask |= AE_READABLE;
            if (state->events[i].events & EPOLLOUT) mask |= AE_WRITABLE;
            eventLoop->fired[i].fd = state->events[i].data.fd;
            eventLoop->fired[i].mask = mask;
        }
    }
    return numevents;
}
```

## 3. 时间事件

### 3.1 创建时间事件

```c
/* 创建定时器 */
long long aeCreateTimeEvent(aeEventLoop *eventLoop, long long milliseconds,
                            aeTimeProc *proc, void *clientData,
                            aeEventFinalizerProc *finalizerProc)
{
    long long id = eventLoop->timeEventNextId++;
    aeTimeEvent *te = zmalloc(sizeof(*te));

    te->id = id;
    te->clientData = clientData;
    te->timeProc = proc;
    te->finalizerProc = finalizerProc;
    te->refcount = 0;

    /* 计算触发时间 */
    monotime now = getMonotonicUs();
    te->when = sumMonotonicUs(now, milliseconds * 1000);

    /* 插入到有序链表 */
    te->next = eventLoop->timeEventHead;
    if (eventLoop->timeEventHead) {
        eventLoop->timeEventHead->prev = te;
    }
    eventLoop->timeEventHead = te;
    te->prev = NULL;

    return id;
}

/* 删除定时器 */
int aeDeleteTimeEvent(aeEventLoop *eventLoop, long long id) {
    aeTimeEvent *te = eventLoop->timeEventHead;
    while(te) {
        if (te->id == id) {
            te->id = AE_DELETED_EVENT_ID;
            return C_OK;
        }
        te = te->next;
    }
    return C_ERR;
}
```

### 3.2 时间事件处理

```c
/* 处理到期的时间事件 */
int aeProcessTimeEvents(aeEventLoop *eventLoop) {
    int processed = 0;
    aeTimeEvent *te, *prev;

    te = eventLoop->timeEventHead;
    while(te) {
        now = getMonotonicUs();
        if (te->when <= now) {
            int retval;

            te->refcount++;
            retval = te->timeProc(eventLoop, te->id, te->clientData);
            te->refcount--;

            processed++;

            if (retval == AE_NOMORE) {
                /* 删除事件 */
                te->id = AE_DELETED_EVENT_ID;
            }

            prev = te;
            te = te->next;
        } else {
            break;
        }
    }

    return processed;
}
```

### 3.3 Redis 中的时间事件

```c
/* serverCron - 核心定时任务 */
int serverCron(aeEventLoop *eventLoop, long long id, void *clientData) {
    /* 运行频率：10Hz (每秒 10 次) */

    /* 1. 软件时钟 */
    server.tick++;

    /* 2. 时间更新 */
    updateTime();

    /* 3. 检查 RDB/AOF 保存条件 */
    if (server.rdb_save_time_last > 0) {
        /* 检查保存条件 */
    }

    /* 4. 处理过期键 */
    if (server.active_expire_enabled) {
        activeExpireCycle(ACTIVE_EXPIRE_CYCLE_SLOW);
    }

    /* 5. 更新统计信息 */
    trackInstantaneousMetric(STATS_METRIC_COMMAND, server.stat_numcommands);

    /* 6. 复制任务 */
    replicationCron();

    /* 7. 集群任务 */
    if (server.cluster_enabled) {
        clusterCron();
    }

    /* 8. 内存管理 */
    if (server.maxmemory) {
        freeMemoryIfNeeded();
    }

    return 1000 / CONFIG_DEFAULT_HZ;  /* 下次调用间隔 (毫秒) */
}
```

## 4. 文件事件处理

### 4.1 客户端连接处理

```c
/* 接受新连接 */
void acceptTcpHandler(aeEventLoop *el, int fd, void *privdata, int mask) {
    int cport, cfd, max = MAX_ACCEPTS_PER_CALL;
    char cip[NET_IP_STR_LEN];
    client *c;

    while(max--) {
        cfd = anetTcpAccept(server.neterr, fd, cip, sizeof(cip), &cport);
        if (cfd == ANET_ERR) {
            if (errno != EWOULDBLOCK)
                serverLog(LL_WARNING, "Accepting client connection: %s",
                          server.neterr);
            return;
        }

        /* 创建客户端 */
        c = createClient(cfd);
        if (!c) {
            close(cfd);
            continue;
        }

        serverLog(LL_VERBOSE,"Accepted %s:%d", cip, cport);
    }
}

/* 注册接受处理器 */
aeCreateFileEvent(server.el, server.fd[0], AE_READABLE,
                  acceptTcpHandler, NULL);
```

### 4.2 读取命令

```c
/* 读取客户端请求 */
void readQueryFromClient(connection *conn) {
    client *c = connGetPrivateData(conn);
    size_t qblen = sdslen(c->querybuf);

    /* 调整查询缓冲区 */
    if (c->querybuf_peak < qblen) c->querybuf_peak = qblen;
    c->qb_pos = 0;

    /* 读取数据 */
    nread = connRead(conn, c->querybuf + qblen, readlen);
    if (nread == -1) {
        if (connGetState(conn) == CONN_STATE_CONNECTED) {
            return;  /* EAGAIN */
        } else {
            freeClient(c);
            return;
        }
    }

    /* 处理输入缓冲区 */
    processInputBuffer(c);
}
```

### 4.3 写入响应

```c
/* 写入客户端响应 */
void writeToClient(client *c) {
    ssize_t nwritten = 0, totwritten = 0;

    while(clientHasPendingReplies(c)) {
        if (c->bufpos > 0) {
            /* 写入固定缓冲区 */
            nwritten = connWrite(c->conn, c->buf + c->sentlen, c->bufpos - c->sentlen);
        } else {
            /* 写入链式缓冲区 */
            robj *o = listNodeValue(listFirst(c->reply));
            nwritten = connWrite(c->conn, o->ptr + c->sentlen,
                                sdslen(o->ptr) - c->sentlen);
        }

        if (nwritten <= 0) break;

        c->sentlen += nwritten;
        totwritten += nwritten;

        /* 检查是否完成 */
        if (c->bufpos > 0 && c->sentlen == c->bufpos) {
            c->bufpos = 0;
            c->sentlen = 0;
        }
    }

    if (totwritten > 0) {
        c->lastinteraction = server.unixtime;
    }

    if (!clientHasPendingReplies(c)) {
        /* 完成写入，禁用可写事件 */
        aeDeleteFileEvent(server.el, c->fd, AE_WRITABLE);
    }
}
```

## 5. IO 多线程 (Redis 6.0+)

### 5.1 架构设计

```
主线程                           IO 线程 1         IO 线程 2
  |                                 |                 |
  |-- 接受连接 -------------------> |                 |
  |                                 |                 |
  |  分配客户端到 IO 线程 ----------> | 读取请求        |
  |                                 | 解析协议        |
  |                                 |                 |
  |<------------------------------ | 返回            |
  |                                 |                 |
  |  执行命令 (主线程)              |                 |
  |                                 |                 |
  |  分配响应到 IO 线程 ------------> | 写入响应        |
  |                                 |                 |
```

### 5.2 IO 线程配置

```c
/* 服务器中的 IO 线程配置 */
struct redisServer {
    int io_threads_num;              /* IO 线程数量 */
    pthread_t io_threads[IO_THREADS_MAX_NUM];
    list *io_threads_clients[IO_THREADS_MAX_NUM];  /* 每个线程的客户端队列 */
    atomic_t io_threads_pending;     /* 待处理的 IO 操作数 */
    int io_threads_active;           /* IO 线程是否活跃 */
};
```

```conf

io-threads 4              # IO 线程数量 (默认关闭)
io-threads-do-reads yes   # 是否处理读取 (默认只处理写入)
```

### 5.3 IO 线程处理

```c
/* IO 线程主函数 */
void *ioThreadMain(void *arg) {
    while(1) {
        /* 等待任务 */
        pthread_mutex_lock(&io_threads_mutex);
        while (atomicGet(server.io_threads_pending) == 0) {
            pthread_cond_wait(&io_threads_condition, &io_threads_mutex);
        }
        pthread_mutex_unlock(&io_threads_mutex);

        /* 处理分配的客户端 */
        handleClientsWithPendingReadsUsingThreads();
        handleClientsWithPendingWritesUsingThreads();

        atomicDecr(server.io_threads_pending, 1);
    }
}

/* 使用 IO 线程读取 */
int readQueryFromClientInThread(client *c) {
    /* 在线程中读取 */
    nread = connRead(c->conn, c->querybuf + qblen, readlen);

    /* 放回主线程处理 */
    return C_OK;
}
```

## 6. 关键钩子函数

### 6.1 beforeSleep 回调

```c
/* 每次事件循环睡眠前调用 */
void beforeSleep(aeEventLoop *eventLoop) {
    /* 1. 处理复制 ACK */
    if (server.masterhost == NULL) {
        flushSlavesOutputBuffers();
    }

    /* 2. 处理阻塞客户端 */
    handleBlockedClients();

    /* 3. AOF fsync */
    if (server.aof_state != AOF_OFF) {
        flushAppendOnlyFile(0);
    }

    /* 4. 更新统计 */
    trackInstantaneousMetric(STATS_METRIC_EL_CYCLE, 1);
    trackInstantaneousMetric(STATS_METRIC_EL_DURATION,
                             elapsedUs(el_start));
}
```

### 6.2 afterSleep 回调

```c
/* 每次事件循环唤醒后调用 */
void afterSleep(aeEventLoop *eventLoop) {
    /* 集群处理 */
    if (server.cluster_enabled) {
        clusterBeforeSleep();
    }
}
```

### 7.2 Redis 集成

```c
/* Redis 服务器初始化 */
void initServer(void) {
    /* 创建事件循环 */
    server.el = aeCreateEventLoop(server.maxclients + CONFIG_FDSET_INCR);

    /* 注册接受处理器 */
    for (j = 0; j < server.ipfd_count; j++) {
        aeCreateFileEvent(server.el, server.fd[j], AE_READABLE,
                          acceptTcpHandler, NULL);
    }

    /* 注册时间事件 */
    aeCreateTimeEvent(server.el, 1, serverCron, NULL, NULL);

    /* 设置钩子 */
    aeSetBeforeSleepProc(server.el, beforeSleep);
    aeSetAfterSleepProc(server.el, afterSleep);
}
```
